# KB Reviews (S10) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Converge two non-conforming KB review list endpoints into one cursor-paginated, visibility-gated list; eliminate the persisted `expired` status; add bulk decide with partial-success; and rebuild the frontend Reviews page on the shared kit.

**Architecture:** The `status=overdue` filter is computed at read time (`pending AND dueAt < now()`) and exposed as a derived boolean field on each row — never stored. The two existing list endpoints (`/kb/page-reviews` and `/kb/page-reviews/due`) converge to one service method; the `/due` path is kept because the e2e spec references it. Bulk decide is per-row result, idempotent, max 100 ids, rejecting without a reason at the Zod schema level. The frontend page is rebuilt on `usePageState`/`<PageState>`, cursor-paginated `DataTable`, `useUrlFilters` for URL-backed filters, and a bulk decide action that reports partial success.

**Tech Stack:** NestJS 11.1 · Drizzle 0.45 · Zod 4 · cursor pagination from `common/pagination/cursor.ts` · `keysetAfterId` from `common/pagination/keyset.ts` · `buildCursorPage` · `cursorPageSchema` from `common/openapi/response-envelopes.ts` · `KnowledgeAuthorizationService.visiblePagePredicate` · React 19.2 · TanStack Query 5.90 · `useCursorPagination` · `useUrlFilters` · `usePageState` · `<PageState>` · `<DataTable pagination={{ mode: "cursor" }}>`.

**Spec:** `docs/specs/knowledge-base/02-page-component-spec.md` §8 (Reviews — `/knowledge/wiki/reviews`).

```typescript
import { ForbiddenException, Inject, Injectable } from "@nestjs/common";
import { and, asc, desc, eq, gte, isNull, lt, lte, or, sql } from "drizzle-orm";
import { alias } from "drizzle-orm/pg-core";
import {
  kbPageReviews,
  kbPages,
  kbSpaces,
  organizationMembers,
  users,
} from "../../../db/schema";
import { DRIZZLE } from "../../../db/drizzle.constants";
import { type Db } from "../../../db/drizzle.module";
import type { CurrentUserContext } from "../../../common/auth/backend-claims";
import { AccessService } from "../../access/access.service";
import { actingMembershipId } from "../../../common/auth/principal";
import { keysetAfterId } from "../../../common/pagination/keyset";
import { buildCursorPage, decodeCursor, type CursorPage } from "../../../common/pagination/cursor";
import { reviewerCanSeeAllReviews } from "./kb-page-reviews.service";
import { KnowledgeAuthorizationService } from "../core/authorization/knowledge-authorization.service";
import type { ListPageReviewsQuery } from "./dto/kb-page-reviews.schemas";

type ReviewRow = typeof kbPageReviews.$inferSelect;

export interface ReviewListItem {
  id: number;
  orgId: string;
  pageId: number;
  type: ReviewRow["type"];
  status: ReviewRow["status"];
  isOverdue: boolean;
  requestedById: string | null;
  reviewerId: string | null;
  requestedByMembershipId: number | null;
  reviewerMembershipId: number | null;
  dueAt: Date | null;
  decidedAt: Date | null;
  decisionNote: string | null;
  createdAt: Date;
  updatedAt: Date;
  pageTitle: string | null;
  requestedByName: string | null;
  reviewerName: string | null;
}

const REVIEW_LIST_COLUMNS = {
  id: kbPageReviews.id,
  orgId: kbPageReviews.orgId,
  pageId: kbPageReviews.pageId,
  type: kbPageReviews.type,
  status: kbPageReviews.status,
  requestedById: kbPageReviews.requestedById,
  reviewerId: kbPageReviews.reviewerId,
  requestedByMembershipId: kbPageReviews.requestedByMembershipId,
  reviewerMembershipId: kbPageReviews.reviewerMembershipId,
  dueAt: kbPageReviews.dueAt,
  decidedAt: kbPageReviews.decidedAt,
  decisionNote: kbPageReviews.decisionNote,
  createdAt: kbPageReviews.createdAt,
  updatedAt: kbPageReviews.updatedAt,
};

@Injectable()
export class KbPageReviewsQueryService {
  constructor(
    @Inject(DRIZZLE) private readonly db: Db,
    private readonly access: AccessService,
    private readonly auth: KnowledgeAuthorizationService,
  ) {}

  private actorMembershipId(user: CurrentUserContext): number {
    const membershipId = actingMembershipId(user.principal);
    if (membershipId === null)
      throw new ForbiddenException("An organization membership is required");
    return membershipId;
  }

  async list(user: CurrentUserContext, query: ListPageReviewsQuery): Promise<CursorPage<ReviewListItem>> {
    const requester = alias(users, "requester");
    const reviewer = alias(users, "reviewer");
    const requesterMembership = alias(organizationMembers, "reviewer_requester_membership");
    const reviewerMembership = alias(organizationMembers, "reviewer_assignee_membership");

    const visibilityPredicate = await this.auth.visiblePagePredicate(user, "view");
    const canSeeAll = await reviewerCanSeeAllReviews(user, this.access);
    const now = new Date();

    const conditions: ReturnType<typeof eq>[] = [
      eq(kbPageReviews.orgId, user.orgId) as ReturnType<typeof eq>,
    ];

    if (query.status === "overdue") {
      conditions.push(eq(kbPageReviews.status, "pending") as ReturnType<typeof eq>);
      conditions.push(lt(kbPageReviews.dueAt, now) as ReturnType<typeof eq>);
    } else if (query.status !== undefined) {
      conditions.push(eq(kbPageReviews.status, query.status) as ReturnType<typeof eq>);
    }

    if (query.type !== undefined) {
      conditions.push(eq(kbPageReviews.type, query.type) as ReturnType<typeof eq>);
    }

    if (query.spaceId !== undefined) {
      conditions.push(eq(kbPages.spaceId, query.spaceId) as ReturnType<typeof eq>);
    }

    if (query.dueFrom !== undefined) {
      conditions.push(gte(kbPageReviews.dueAt, new Date(query.dueFrom)) as ReturnType<typeof eq>);
    }

    if (query.dueTo !== undefined) {
      conditions.push(lte(kbPageReviews.dueAt, new Date(query.dueTo)) as ReturnType<typeof eq>);
    }

    if (!canSeeAll) {
      const membershipId = this.actorMembershipId(user);
      const ownOnly = or(
        eq(kbPageReviews.reviewerMembershipId, membershipId),
        eq(kbPageReviews.requestedByMembershipId, membershipId),
      );
      if (ownOnly) conditions.push(ownOnly as ReturnType<typeof eq>);
    }

    const position = decodeCursor(query.cursor);
    if (position) {
      conditions.push(
        keysetAfterId(kbPageReviews.dueAt, kbPageReviews.id, {
          sortValue: position.sortValue,
          id: position.id,
        }) as ReturnType<typeof eq>,
      );
    }

    const orderFn = query.sortDir === "desc" ? desc : asc;

    const rows = await this.db
      .select({
        ...REVIEW_LIST_COLUMNS,
        pageTitle: kbPages.title,
        requestedByName: requester.name,
        reviewerName: reviewer.name,
      })
      .from(kbPageReviews)
      .innerJoin(
        kbPages,
        and(
          eq(kbPageReviews.pageId, kbPages.id),
          eq(kbPageReviews.orgId, kbPages.orgId),
          isNull(kbPages.deletedAt),
          visibilityPredicate,
        ),
      )
      .leftJoin(requesterMembership, eq(kbPageReviews.requestedByMembershipId, requesterMembership.id))
      .leftJoin(reviewerMembership, eq(kbPageReviews.reviewerMembershipId, reviewerMembership.id))
      .leftJoin(requester, eq(requesterMembership.userId, requester.id))
      .leftJoin(reviewer, eq(reviewerMembership.userId, reviewer.id))
      .where(and(...conditions))
      .orderBy(orderFn(kbPageReviews.dueAt), orderFn(kbPageReviews.id))
      .limit(query.limit + 1);

    return buildCursorPage(
      rows.map((row) => ({
        ...row,
        isOverdue: row.status === "pending" && row.dueAt !== null && row.dueAt < now,
      })),
      query.limit,
      (row) => ({
        sortValue: row.dueAt ? row.dueAt.toISOString() : new Date(0).toISOString(),
        id: String(row.id),
      }),
    );
  }
}
```

**Important notes:**
- The `visibilityPredicate` is applied in the `innerJoin` condition on `kbPages`. This means a review on a hidden page is excluded from results — the reviewer cannot infer that a hidden page has pending reviews from metadata (count, title, due date). This fixes the disclosure bug.
- `isOverdue` is computed at read time from the fetched rows, never stored.
- `spaceId` filter joins via `kbPages.spaceId` — the `innerJoin` already brings `kbPages` in.
- The `reviewer` filter (by userId) is a query-param filter but looking at the types — `query.reviewer` is a userId string. Add it to conditions.

Add the reviewer filter:

```typescript
  conditions.push(eq(kbPageReviews.reviewerId, query.reviewer) as ReturnType<typeof eq>);
}
```

```typescript
    });

    if (!existing) {
      results.push({ id, outcome: "notFound" });
      continue;
    }

    const [visiblePage] = await this.db
      .select({ id: kbPages.id })
      .from(kbPages)
      .where(
        and(
          eq(kbPages.id, existing.pageId),
          eq(kbPages.orgId, user.orgId),
          isNull(kbPages.deletedAt),
          visibilityPredicate,
        ),
      )
      .limit(1);

    if (!visiblePage) {
      results.push({ id, outcome: "notFound" });
      continue;
    }

    if (existing.status !== "pending") {
      results.push({ id, outcome: "conflict" });
      continue;
    }

    await this.db
      .update(kbPageReviews)
      .set({
        status: input.decision,
        reviewerId: user.userId,
        reviewerMembershipId,
        decidedAt: new Date(),
        decisionNote: input.note ?? null,
      })
      .where(
        and(
          eq(kbPageReviews.id, id),
          eq(kbPageReviews.orgId, user.orgId),
          eq(kbPageReviews.status, "pending"),
        ),
      );

    this.audit.log({
      action: `kb.review.${input.decision}`,
      userId: user.userId,
      orgId: user.orgId,
      resourceType: "kb_page_review",
      resourceId: String(id),
      before: { status: "pending" },
      after: { status: input.decision },
    });

    results.push({ id, outcome: "succeeded" });
  }

  return { results };
}
```

**Idempotency note:** A review already decided returns `conflict` (not an error). The caller reads the outcome and retries only the pending ones. This is safe and explicit — the brief says "deciding an already-decided review returns its existing outcome rather than erroring". We return `conflict` which communicates "already decided" without leaking existence of hidden reviews (a hidden review returns `notFound`, same as a missing one).

**Visibility note:** `notFound` for both a hidden review and a missing one (the page visibility check and existence check produce the same outcome). This means a caller cannot distinguish whether review id 42 exists but is on a hidden page, or simply doesn't exist.

```tsx
            >
              Approve
            </Button>
            <Button
              type="button"
              size="sm"
              variant="outline"
              className="text-xs text-destructive border-destructive/30 hover:bg-destructive/10"
              onClick={makeRejectHandler(review)}
            >
              Reject
            </Button>
          </div>
        );
      },
    },
  ], [canManage]);

  const filters = (
    <>
      <Select value={statusFilter} onValueChange={handleStatusChange}>
        <SelectTrigger className="w-[130px]">
          <SelectValue placeholder="Status" />
        </SelectTrigger>
        <SelectContent>
          <SelectItem value="all">All statuses</SelectItem>
          <SelectItem value="pending">Pending</SelectItem>
          <SelectItem value="approved">Approved</SelectItem>
          <SelectItem value="rejected">Rejected</SelectItem>
          <SelectItem value="overdue">Overdue</SelectItem>
        </SelectContent>
      </Select>
      <Select value={typeFilter} onValueChange={handleTypeChange}>
        <SelectTrigger className="w-[130px]">
          <SelectValue placeholder="Type" />
        </SelectTrigger>
        <SelectContent>
          <SelectItem value="all">All types</SelectItem>
          <SelectItem value="approval">Approval</SelectItem>
          <SelectItem value="freshness">Freshness</SelectItem>
        </SelectContent>
      </Select>
      {canManage && selectedIds.size > 0 && (
        <Button
          type="button"
          size="sm"
          variant="outline"
          onClick={() => setShowBulkDecide(true)}
        >
          Decide {selectedIds.size} selected
        </Button>
      )}
    </>
  );

  return (
    <>
      {approveTarget && (
        <ApproveDialog review={approveTarget} onClose={handleCloseApprove} />
      )}
      {rejectTarget && (
        <RejectDialog review={rejectTarget} onClose={handleCloseReject} />
      )}
      {showBulkDecide && (
        <BulkDecideDialog
          selectedIds={Array.from(selectedIds)}
          reviews={reviews}
          onClose={handleCloseBulk}
          onSuccess={handleBulkSuccess}
        />
      )}
      <PageWrapper title="Reviews" filters={filters}>
        <PageState
          resolution={pageState}
          loading={<ReviewsTableSkeleton />}
          onRetry={refetch}
          empty={
            <div className="flex flex-1 items-center justify-center p-8">
              <div className="text-center space-y-2">
                <KbClipboardCheckIcon className="w-8 text-muted-foreground mx-auto" />
                <p className="text-sm font-medium">No reviews</p>
                <p className="text-sm text-muted-foreground">
                  {statusFilter !== "all" || typeFilter !== "all"
                    ? "No reviews match the current filters."
                    : "No pages are currently under review."}
                </p>
              </div>
            </div>
          }
        >
          <DataTable
            data={reviews}
            columns={columns}
            getRowKey={(review) => review.id}
            isLoading={isLoading}
            className="flex-1 min-h-0"
            selection={
              canManage
                ? {
                    selected: new Set(Array.from(selectedIds).map(String)),
                    onChange: handleSelectionChange,
                    isRowSelectable: (review) => review.status === "pending",
                    getRowLabel: (review) => review.pageTitle ?? `Review #${review.id}`,
                  }
                : undefined
            }
            pagination={{
              mode: "cursor",
              pageSize: 50,
              pageNumber: pagination.pageNumber,
              hasMore: data?.pagination.hasMore ?? false,
              hasPrevious: pagination.hasPrevious,
              onNext: () => pagination.goNext(data?.pagination.nextCursor),
              onPrevious: pagination.goPrevious,
            }}
          />
        </PageState>
      </PageWrapper>
    </>
  );
}
```
